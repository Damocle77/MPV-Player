# 1. Fix Boost-mageddon (le runtime vitali)
ubuntu 24:
apt update && apt install -y libboost-coroutine1.83.0 libboost-context1.83.0 libboost-filesystem1.83.0 libboost-iostreams1.83.0 libboost-thread1.83.0 libboost-program-options1.83.0

ubuntu 22:
apt-get install -y icinga2 icinga2-bin icinga2-common


# 2. Installa i pacchetti icinga 
dpkg -i icinga2* 


# 3. Rinfrescata al linker e start del demone
ldconfig
systemctl enable --now icinga2
 
# 4. Test manuale e validazione finale
chmod -R +x /usr/lib/nagios/plugins && chmod +x /usr/lib/nagios/plugins
/usr/lib/nagios/plugins/check_load -w 5,4,3 -c 10,8,6
icinga2 daemon -C && systemctl restart icinga2
